function loadFonts() { return new Promise((resolve) => { if (!document.fonts) { setTimeout(resolve, 800) } else { document.fonts.ready.then(resolve) } }) }

Module Usage

How to use the Fontloader module in your application.

import { Application, Fontloader } from 'jupiter' const app = new Application() const fontloader = new Fontloader(app) fontloader.loadFonts().then(() => { console.log('Fonts are ready!') // Your code here })
